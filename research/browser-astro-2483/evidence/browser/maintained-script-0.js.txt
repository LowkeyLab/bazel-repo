
      (() => {
        function preferredTheme() {
          try {
            const stored = localStorage.getItem("theme");
            if (stored === "light" || stored === "dracula") return stored;
          } catch {
            // Fall back to the system preference if storage is unavailable.
          }
          return matchMedia("(prefers-color-scheme: dark)").matches ? "dracula" : "light";
        }
        document.documentElement.dataset.theme = preferredTheme();
        document.addEventListener("astro:before-swap", (event) => {
          event.newDocument.documentElement.dataset.theme = document.documentElement.dataset.theme;
        });
      })();
